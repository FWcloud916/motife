import { Graph, layout as dagreLayout } from "@dagrejs/dagre";
import { buildRoundedPath } from "./edgePath";
import { NODE_SIZE } from "./nodeSizing";
import type { NodeSize } from "./nodeSizing";
import type { GraphSpec, LayoutEdgePath, LayoutRect, LayoutResult } from "./types";
 
const NODE_SEP = 64; // tokens.spacing.lg — kept as a literal to avoid a
const RANK_SEP = 96; // circular import between layout/ and tokens/index.ts
 
interface DagreNodeLabel {
  width: number;
  height: number;
  x: number;
  y: number;
}
 
interface DagreEdgeLabel {
  points: { x: number; y: number }[];
}
 
interface DagreGraphLabel {
  width?: number;
  height?: number;
}
 
/**
 * The one place topology becomes coordinates (motife-plan.md §2 決策3:
 * the LLM/caller only ever describes A→B, never a position). Wrapped
 * behind this function signature — not the dagre call site directly — so
 * swapping the engine (e.g. to ELK for orthogonal routing) later is a
 * one-file change; every other component only ever sees a LayoutResult.
 *
 * Synchronous by design: Remotion renders a frame synchronously, and
 * elkjs's Promise-only API would force delayRender() plumbing inside every
 * Diagram instance. dagre's rank-based layout is also exactly the shape
 * explainer flow diagrams need.
 *
 * `nodeSizes` optionally overrides a node's footprint by id — that's how
 * text-measured widths reach the layout (see measureNodes.ts) without
 * putting dimensions into GraphSpec, which must stay purely semantic
 * (motife-plan.md §2 決策3: the DSL describes topology, never geometry).
 * A plain record rather than a callback keeps this function serializable
 * in spirit, trivially fakeable in node tests, and gives Diagram's useMemo
 * a value to depend on instead of a function identity. Omit it — as every
 * unit test does — and every node falls back to its Size token.
 */
export function computeLayout(
  graph: GraphSpec,
  nodeSizes?: Record<string, NodeSize>,
): LayoutResult {
  // dagre's own TS types re-export @dagrejs/graphlib's `Graph<G, N, E>`
  // generically as `any` by default — the cast below narrows to the
  // label shapes this function actually reads/writes.
  const g = new Graph() as unknown as {
    setGraph: (label: { rankdir: string; nodesep: number; ranksep: number }) => void;
    setDefaultEdgeLabel: (fn: () => Record<string, never>) => void;
    setNode: (id: string, label: { width: number; height: number }) => void;
    setEdge: (from: string, to: string, label: { label?: string }) => void;
    nodes: () => string[];
    node: (id: string) => DagreNodeLabel;
    edges: () => { v: string; w: string }[];
    edge: (edge: { v: string; w: string } | string, w?: string) => DagreEdgeLabel;
    graph: () => DagreGraphLabel;
  };
 
  g.setGraph({
    rankdir: graph.direction === "down" ? "TB" : "LR",
    nodesep: NODE_SEP,
    ranksep: RANK_SEP,
  });
  g.setDefaultEdgeLabel(() => ({}));
 
  for (const node of graph.nodes) {
    const { width, height } = nodeSizes?.[node.id] ?? NODE_SIZE[node.size ?? "md"];
    g.setNode(node.id, { width, height });
  }
 
  const edgeIds = new Map<string, string>();
  for (const edge of graph.edges) {
    const id = edge.id ?? `${edge.from}->${edge.to}`;
    g.setEdge(edge.from, edge.to, { label: edge.label });
    edgeIds.set(`${edge.from}${edge.to}`, id);
  }
 
  dagreLayout(g as never);
 
  const nodes: Record<string, LayoutRect> = {};
  for (const id of g.nodes()) {
    const n = g.node(id);
    nodes[id] = {
      // dagre reports node centers; every LayoutRect in the library is
      // top-left, matching CSS absolute positioning.
      x: n.x - n.width / 2,
      y: n.y - n.height / 2,
      width: n.width,
      height: n.height,
    };
  }
 
  const edges: Record<string, LayoutEdgePath> = {};
  for (const e of g.edges()) {
    const data = g.edge(e);
    const points = data.points ?? [];
    const id = edgeIds.get(`${e.v}${e.w}`) ?? `${e.v}->${e.w}`;
    edges[id] = { points, path: buildRoundedPath(points) };
  }
 
  const graphLabel = g.graph();
  return {
    width: graphLabel.width ?? 0,
    height: graphLabel.height ?? 0,
    nodes,
    edges,
  };
}
